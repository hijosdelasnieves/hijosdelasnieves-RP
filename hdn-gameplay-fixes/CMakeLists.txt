cmake_minimum_required(VERSION 3.22)
project(HdnGameplayFixes VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HDN_BUILD_PLUGIN "Build Windows x64 SKSE plugin" ${WIN32})
enable_testing()
add_executable(gameplay-policy-tests tests/PolicyTests.cpp)
target_include_directories(gameplay-policy-tests PRIVATE include)
target_compile_options(gameplay-policy-tests PRIVATE
  $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>
  $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
add_test(NAME gameplay-policy COMMAND gameplay-policy-tests)
if(HDN_BUILD_PLUGIN)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Plugin requires MSVC x64")
  endif()
  include(FetchContent)
  set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(commonlib
    GIT_REPOSITORY https://github.com/alandtse/CommonLibSSE-NG.git
    GIT_TAG 94faaed0c60eddd8347767f2d4d29a97c93bde8c)
  FetchContent_MakeAvailable(commonlib)
  get_target_property(HDN_COMMONLIB_IMPORTED CommonLibSSE IMPORTED)
  if(NOT HDN_COMMONLIB_IMPORTED)
    # MSBuild --parallel schedules projects; /MP parallelizes this one library.
    target_compile_options(CommonLibSSE PRIVATE /MP4)
  endif()
  include(${commonlib_SOURCE_DIR}/cmake/CommonLibSSE.cmake)
  set(HDN_RUNTIMES 1.6.1170)
  foreach(index RANGE 1 15)
    list(APPEND HDN_RUNTIMES 0.0.0)
  endforeach()
  add_commonlibsse_plugin(HdnGameplayFixes AUTHOR "Hijos de las Nieves"
    STRUCT_DEPENDENT COMPATIBLE_RUNTIMES ${HDN_RUNTIMES}
    SOURCES src/Plugin.cpp)
  target_include_directories(HdnGameplayFixes PRIVATE include)
  target_compile_definitions(HdnGameplayFixes PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(HdnGameplayFixes PRIVATE Xinput9_1_0)
  target_precompile_headers(HdnGameplayFixes PRIVATE include/PCH.hpp)
  set_target_properties(HdnGameplayFixes PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/package/Data/SKSE/Plugins")
  add_executable(gameplay-loader-tests tests/LoaderTests.cpp)
  add_dependencies(gameplay-loader-tests HdnGameplayFixes)
  add_test(NAME gameplay-loader-disabled COMMAND gameplay-loader-tests "$<TARGET_FILE:HdnGameplayFixes>" disabled)
  add_test(NAME gameplay-loader-missing-library COMMAND gameplay-loader-tests "$<TARGET_FILE:HdnGameplayFixes>" missing-library)
endif()
